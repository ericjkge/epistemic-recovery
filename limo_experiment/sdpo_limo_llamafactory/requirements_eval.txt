# Evaluation environment — minimal deps for vLLM inference
#
# Install into a SEPARATE virtual environment from training (see README.md).
# vLLM pins its own versions of torch and transformers; mixing with the
# training environment causes import errors.
#
# Setup (one-time):
#   python3.11 -m venv .venv-eval && source .venv-eval/bin/activate
#   pip install -r requirements_eval.txt
#
# Pick ONE vLLM version line below based on your hardware:

# H100 / A100 / GH200 (CUDA 12.x)
# Note: 0.8.4 has a LoRA/Punica crash (see README troubleshooting); 0.8.5.post1 fixes it.
vllm==0.8.5.post1

# Blackwell (B100 / B200 / RTX 50 series) — uncomment and comment line above
# vllm>=0.12.0

# Pin transformers: vLLM 0.8.5.post1's loose bound otherwise resolves to transformers>=5.0,
# which removed `all_special_tokens_extended` and breaks vLLM's tokenizer init.
transformers==4.51.3

# tqdm is imported by eval/evaluate_aime.py directly (HF backend path)
tqdm

# LoRA adapter loading / merging
peft>=0.14.0

# AIME benchmark loading + HuggingFace model downloads
datasets>=2.14.0
huggingface_hub>=0.23.0

# Plotting (used by analyze_epistemic.py, plot_probe_curves.py,
# analyze_token_distribution.py, evaluate_epistemic_alignment.py)
matplotlib>=3.7.0

# Symbolic-equivalence grader for MATH500 (rule-lighteval/MATH_v2 style).
# Not needed for AIME-only runs; loaded lazily by eval/evaluate_aime.py.
math-verify
